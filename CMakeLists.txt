cmake_minimum_required(VERSION 3.16)
project(EternalRPG VERSION 0.1.0 LANGUAGES CXX)
option(ER_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
add_library(eternal_core src/renderer/framebuffer.cpp src/platform/terminal.cpp src/world/world.cpp src/game.cpp src/ascii_conversion/ascii.cpp)
target_sources(eternal_core PRIVATE src/ui/hud.cpp src/ui/menus.cpp src/saves.cpp src/worldgen/atlas.cpp src/ui/map.cpp src/interiors/interior.cpp src/interiors/travel.cpp src/npc/population.cpp src/history/chronicle.cpp src/dialogue/social.cpp src/economy/trade.cpp src/history/persistence.cpp src/worldgen/environment.cpp src/progression/progression.cpp src/factions/crime.cpp src/quests/quest.cpp)
target_sources(eternal_core PRIVATE src/assets/data.cpp src/combat/combat.cpp src/mods/mods.cpp)
target_sources(eternal_core PRIVATE src/ai/ai.cpp)
target_sources(eternal_core PRIVATE src/ai/party.cpp src/ai/dragon.cpp)
target_compile_definitions(eternal_core PUBLIC ER_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")
target_include_directories(eternal_core PUBLIC src)
if(MSVC)
  target_compile_options(eternal_core PUBLIC /W4 /permissive- /utf-8)
else()
  target_compile_options(eternal_core PUBLIC -Wall -Wextra -Wpedantic)
  if(ER_SANITIZE)
    target_compile_options(eternal_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(eternal_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(eternal_rpg src/main.cpp)
if(APPLE)
  enable_language(OBJCXX)
  set_property(TARGET eternal_rpg PROPERTY OBJCXX_STANDARD 17)
  target_sources(eternal_rpg PRIVATE src/platform/window.mm)
  set_source_files_properties(src/platform/window.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
  target_link_libraries(eternal_rpg PRIVATE "-framework AppKit" "-framework Metal" "-framework QuartzCore" "-framework CoreText")
endif()
target_link_libraries(eternal_rpg PRIVATE eternal_core)
if(WIN32)
  target_compile_definitions(eternal_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
enable_testing()
add_executable(eternal_tests tests/tests.cpp)
target_link_libraries(eternal_tests PRIVATE eternal_core)
add_test(NAME unit COMMAND eternal_tests)

if(APPLE)
  # Run manually in a graphical session; regular CTest remains headless.
  add_executable(eternal_native_input_tests tests/native_input.mm src/platform/window.mm)
  set_property(TARGET eternal_native_input_tests PROPERTY OBJCXX_STANDARD 17)
  target_compile_options(eternal_native_input_tests PRIVATE -fobjc-arc)
  target_link_libraries(eternal_native_input_tests PRIVATE eternal_core "-framework AppKit" "-framework Metal" "-framework QuartzCore" "-framework CoreText")
endif()
