cmake_minimum_required(VERSION 3.16)
project(EternalRPG VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
add_library(eternal_core src/renderer/framebuffer.cpp src/platform/terminal.cpp src/world/world.cpp src/game.cpp)
target_include_directories(eternal_core PUBLIC src)
if(MSVC)
  target_compile_options(eternal_core PUBLIC /W4 /permissive- /utf-8)
else()
  target_compile_options(eternal_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(eternal_rpg src/main.cpp)
target_link_libraries(eternal_rpg PRIVATE eternal_core)
enable_testing()
